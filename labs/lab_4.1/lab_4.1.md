8000 78        ; MOV A, B

8002 04

8003 D2        ; JC end (if B < 4) | C_flag = 1
8004 07
8005 80

8006 3E        ; MVI C, FFh (if B >= 4) | ( C_flag = 0)
8007 FF

end:
8008 76        ; HLT
